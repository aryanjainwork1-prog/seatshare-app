export function calculateDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371;
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLon = ((lon2 - lon1) * Math.PI) / 180;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos((lat1 * Math.PI) / 180) * Math.cos((lat2 * Math.PI) / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

export function searchLocations(query: string, userLat = 19.033, userLng = 73.0297, limit = 8) {
  if (!query || query.length < 2) return [];
  const q = query.toLowerCase().trim();
  return mumbaiLocations
    .map(loc => {
      const nameMatch = loc.name.toLowerCase().includes(q) ? 3 : 0;
      const addrMatch = loc.address.toLowerCase().includes(q) ? 2 : 0;
      const matchScore = nameMatch + addrMatch;
      if (matchScore === 0) return null;
      const dist = calculateDistance(userLat, userLng, loc.lat, loc.lng);
      const proximityBonus = Math.max(0, 5 - dist / 3);
      const popularBonus = loc.popular ? 2 : 0;
      const totalScore = matchScore * 10 + proximityBonus + popularBonus;
      return { ...loc, distance: Math.round(dist * 10) / 10, score: Math.round(totalScore) };
    })
    .filter(Boolean)
    .sort((a, b) => (b.score || 0) - (a.score || 0))
    .slice(0, limit);
}

export function findNearestLocation(lat: number, lng: number, maxDistKm = 1.5) {
  let nearest = null;
  let minDist = Infinity;
  for (const loc of mumbaiLocations) {
    const d = calculateDistance(lat, lng, loc.lat, loc.lng);
    if (d < minDist && d <= maxDistKm) {
      minDist = d;
      nearest = loc;
    }
  }
  return nearest;
}

export function estimateFare(distanceKm: number, durationMin = 0) {
  const baseFare = 45;
  const perKm = 14;
  const perMin = 1.5;
  const distFare = Math.round(distanceKm * perKm);
  const timeFare = Math.round((durationMin || distanceKm * 2.5) * perMin);
  const total = Math.max(80, Math.round(baseFare + distFare + timeFare));
  return { fare: total, breakdown: `Base ₹${baseFare} + ₹${perKm}/km + time` };
}
